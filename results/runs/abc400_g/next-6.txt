```python
import sys
import heapq

def select_k_nonadjacent(vals, K):
    m = len(vals)
    if K == 0 or m == 0:
        return 0
    prv = [i - 1 for i in range(m)]
    nxt = [i + 1 for i in range(m)]
    prv[0] = -1
    nxt[m - 1] = -1
    cur = vals[:]
    active = [True] * m
    heap = [(-vals[i], i) for i in range(m)]
    heapq.heapify(heap)
    ans = 0
    for _ in range(K):
        while True:
            neg_v, i = heapq.heappop(heap)
            if active[i] and cur[i] == -neg_v:
                break
        v = -neg_v
        ans += v
        L = prv[i]
        R = nxt[i]
        if L != -1 and R != -1:
            LL = prv[L]
            RR = nxt[R]
            cur[i] = cur[L] + cur[R] - v
            prv[i] = LL
            nxt[i] = RR
            if LL != -1:
                nxt[LL] = i
            if RR != -1:
                prv[RR] = i
            active[L] = False
            active[R] = False
            heapq.heappush(heap, (-cur[i], i))
        elif L != -1:
            LL = prv[L]
            if LL != -1:
                nxt[LL] = -1
            active[L] = False
            active[i] = False
        elif R != -1:
            RR = nxt[R]
            if RR != -1:
                prv[RR] = -1
            active[R] = False
            active[i] = False
        else:
            active[i] = False
    return ans

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    T = int(data[idx]); idx += 1
    out = []
    for _ in range(T):
        N = int(data[idx]); K = int(data[idx+1]); idx += 2
        cakes = []
        for i in range(N):
            x = int(data[idx]); y = int(data[idx+1]); z = int(data[idx+2]); idx += 3
            cakes.append((x, y, z))
        best = 0
        for key in range(3):
            sc = sorted(cakes, key=lambda c: c[key], reverse=True)
            vals = [max(sc[i][0]+sc[i+1][0], sc[i][1]+sc[i+1][1], sc[i][2]+sc[i+1][2]) for i in range(N-1)]

            if a > best:
                best = a
        out.append(best)

solve()
```
